## Diagrama DER

-- 1. Tabela de Funcionários (Professores/Secretaria que enviam avisos)
CREATE TABLE "Funcionario" (
  "id_funcionario" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nome" VARCHAR(100) NOT NULL,
  "perfil" VARCHAR(50) NOT NULL -- Ex: 'Professor', 'Coordenador', 'Administrador'
);

-- 2. Tabela de Módulos / Turmas (Ex: 1º Módulo DS, 2º Módulo Adm, etc.)
CREATE TABLE "Modulo" (
  "id_modulo" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nome" VARCHAR(50) NOT NULL
);

-- 3. Tabela de Alunos
CREATE TABLE "Aluno" (
  "id_aluno" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "nome" VARCHAR(100) NOT NULL,
  "id_modulo" INTEGER NOT NULL, -- Todo aluno pertence obrigatoriamente a um módulo
  CONSTRAINT fk_aluno_modulo FOREIGN KEY ("id_modulo") 
    REFERENCES "Modulo" ("id_modulo") 
    ON DELETE CASCADE
);

-- 4. Tabela de Notificações / Avisos
CREATE TABLE "Notificacao" (
  "id_notificacao" INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "titulo" VARCHAR(150) NOT NULL,
  "mensagem" TEXT NOT NULL,
  "data" TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- Preenche a data/hora automaticamente ao criar
  "status" BOOLEAN DEFAULT FALSE, -- Ex: lido (true) ou não lido (false)
  "id_funcionario" INTEGER NOT NULL, -- Quem enviou
  "id_modulo_destino" INTEGER, -- Destino opcional: se for para a turma toda
  "id_aluno_destino" INTEGER, -- Destino opcional: se for para um aluno específico
  
  -- Relacionamentos (Chaves Estrangeiras)
  CONSTRAINT fk_notificacao_funcionario FOREIGN KEY ("id_funcionario") 
    REFERENCES "Funcionario" ("id_funcionario"),
    
  CONSTRAINT fk_notificacao_modulo FOREIGN KEY ("id_modulo_destino") 
    REFERENCES "Modulo" ("id_modulo") 
    ON DELETE CASCADE,
    
  CONSTRAINT fk_notificacao_aluno FOREIGN KEY ("id_aluno_destino") 
    REFERENCES "Aluno" ("id_aluno") 
    ON DELETE CASCADE
=====================================================================
## Decisões de modelagem

- `Notificacao` tem duas colunas de destino (`id_modulo_destino` e `id_aluno_destino`), nenhuma delas obrigatória — cada notificação preenche só uma, dependendo se o envio é para uma turma inteira ou para um aluno específico.
- `Funcionario` usa um único campo `perfil` para diferenciar professor de coordenador, em vez de duas tabelas separadas.
- `Modulo` foi modelado separado de "turma" (que existe na escola, mas foi deixada fora do escopo deste projeto para simplificar).
